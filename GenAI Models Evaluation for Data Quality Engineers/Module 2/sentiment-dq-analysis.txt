    "def null_report(df, name):\n",
    "    n = df.isna().sum()\n",
    "    out = n[n > 0].sort_values(ascending=False).rename(\"missing\").to_frame()\n",
    "    out[\"missing_%\"] = (out[\"missing\"] / len(df) * 100).round(1)\n",
    "    print(f\"[{name}] rows={len(df)}\")\n",
    "    print(out if len(out) else \"  no missing values\")\n",
    "    print()\n",
    "    return out\n",
    "\n",
